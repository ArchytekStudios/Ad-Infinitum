
im_hermes_gateway = {
	entity = "im_hermes_gateway_entity"
	construction_entity = "im_hermes_gateway_empty_entity" # swap out for im_hermes_gateway_empty_entity
	portrait = "GFX_megastructure_construction_background"
	place_entity_on_planet_plane = no
	build_type = outside_gravity_well
	show_galactic_map_icon = yes
	show_in_outliner = yes
	potential = { clib_is_infinite_empire = yes }
	possible = {
		hidden_trigger = {
			exists = starbase
		}
		custom_tooltip = {
			fail_text = "requires_inside_border"
			is_inside_border = from
		}
		custom_tooltip = {
			fail_text = "only_one_per_system"
			NOT = { has_megastructure = im_hermes_gateway }
		}
		custom_tooltip = {
			fail_text = "blocked_by_pre_ftl_policy"
			if = {
				limit = {
					any_system_planet = {
						exists = owner
						owner = {
							is_primitive = yes
						}
					}
				}
				from = {
					has_policy_flag = interference_aggressive
				}
			}
		}
		custom_tooltip = {
			fail_text = "requires_surveyed_system"
			NOT = {
				any_system_planet = {
					is_surveyed = {
						who = prev.from
						status = no
					}
				}
			}
		}
	}
	prerequisites = { tech_hermes_gateway }
	show_prereqs = yes
	build_time = 1800
	resources = {
		category = megastructures_gateways
		cost = {
			im_infinityenergy = 1500
			im_infinitymatter = 285
		}
		upkeep = {
			im_infinitypoints = 25
			im_infinityenergy = 15
		}
	}
	construction_blocks_and_blocked_by = self_type

	placement_rules = { # scope?
	 	NOR = {
			star = {
				has_planet_flag = im_has_hermes_gateway
			}
			has_star_flag = im_has_hermes_gateway
	 	}
	}

	dismantle_cost = {
		category = megastructures_hyper_relay
		cost = {
			energy = 250
		}
	}

	dismantle_time = 360

	dismantle_potential = {
		always = yes
	}

	dismantle_possible = {
		exists = owner
		owner = { clib_is_infinite_empire = yes }
	}

	on_dismantle_complete = {
		from = {
			add_resource = {
				im_infinityenergy = 1500
				im_infinitymatter = 285
				mult = modifier:megastructure_dismantle_refund_mult
			}
		}
		remove_star_flag = im_has_hermes_gateway
	}


	bypass_type = angel_gateway
	on_build_complete = { # this = system
		activate_gateway = fromfrom
		star = {
			set_planet_flag = im_has_hermes_gateway
		}
		set_star_flag = im_has_hermes_gateway
	}
}
