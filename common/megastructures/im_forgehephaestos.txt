
im_forgehephaestos_base = {
	entity = "im_forgehephaestos_constructionbase_entity"
	construction_entity = "im_forgehephaestos_constructionbase_entity"
	portrait = "GFX_megastructure_construction_background"
	place_entity_on_planet_plane = yes
	scales_with_planet = yes
	build_type = around_planet
	show_galactic_map_icon = yes
	show_in_outliner = yes
	potential = { clib_is_infinite_empire = yes }
	possible = {
		hidden_trigger = {
			exists = starbase
		}
		custom_tooltip = {
			fail_text = "requires_inside_border"
			is_inside_border = from
		}
		custom_tooltip = {
			fail_text = "only_one_per_country"
			exists = owner
			NOT = {
				owner = {
					has_country_flag = im_has_forgehephaestos
				}
			}
		}
		custom_tooltip = {
			fail_text = "requires_black_hole"
			is_star_class = sc_black_hole
		}
		custom_tooltip = {
			fail_text = "requires_surveyed_system"
			NOT = {
				any_system_planet = {
					is_surveyed = {
						who = prev.from
						status = no
					}
				}
			}
		}
		custom_tooltip = {
			fail_text = "requires_no_existing_megastructure"
			has_no_non_gate_megastructure = yes
			NOT = {
				any_system_planet = {
					is_planet_class = pc_cosmogenesis_world
				}
			}
		}
	}
	placement_rules = {
		planet_possible = {
			custom_tooltip = {
				fail_text = "must_build_around_star"
				is_star = yes
			}
		}
	}
	prerequisites = { tech_forgehephaestos }
	show_prereqs = yes
	build_time = 980
	resources = {
		category = megastructures
		cost = {
			im_infinityenergy = 2750
			im_infinitymatter = 530
		}
		upkeep = {
			im_infinitypoints = 10
			im_infinityenergy = 15
		}
	}
	construction_blocks_and_blocked_by = self_type

	dismantle_cost = {
		category = megastructures
		cost = {
			im_infinitymatter = 125
		}
	}

	dismantle_time = 360

	dismantle_potential = {
		always = yes
	}

	dismantle_possible = {
		exists = owner
		owner = { clib_is_infinite_empire = yes }
	}

	on_dismantle_complete = {
		from = {
			add_resource = {
				im_infinityenergy = 1500
				im_infinitymatter = 285
				mult = modifier:megastructure_dismantle_refund_mult
			}
			remove_country_flag = im_has_forgehephaestos
		}
	}

	on_build_complete = {
		from = { set_country_flag = im_has_forgehephaestos }
		every_system_planet = {
			limit = { is_star = no }
			remove_planet = yes
		}
		set_asteroid_belt = {
			radius = 0
		}
	}
}
