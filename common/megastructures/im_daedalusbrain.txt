
im_daedalusbrain_base = {
	entity = "im_daedalusbrain_base_entity"
	construction_entity = "im_daedalusbrain_base_entity"
	portrait = "GFX_megastructure_construction_background"
	place_entity_on_planet_plane = yes
	scales_with_planet = yes
	build_type = around_planet
	show_galactic_map_icon = yes
	show_in_outliner = yes
	potential = { clib_is_infinite_empire = yes NOT = { has_country_flag = im_has_daedalusbrain } }
	upgrade_from = { im_dysonsphere }
	possible = {
		hidden_trigger = {
			exists = starbase
		}
		custom_tooltip = {
			fail_text = "requires_inside_border"
			is_inside_border = from
		}
		custom_tooltip = {
			fail_text = "requires_no_habitable_planets"
			NOT = {
				any_system_planet = {
					colonizable_planet = yes
				}
			}
		}
		custom_tooltip = {
			fail_text = "requires_searing_star"
			is_star_class = sc_im_searing_star
		}
		custom_tooltip = {
			fail_text = "requires_surveyed_system"
			NOT = {
				any_system_planet = {
					is_surveyed = {
						who = prev.from
						status = no
					}
				}
			}
		}
		custom_tooltip = {
			fail_text = "requires_no_existing_megastructure"
			has_no_non_gate_megastructure = yes
			NOT = {
				any_system_planet = {
					is_planet_class = pc_cosmogenesis_world
				}
			}
		}
	}
	placement_rules = {
		planet_possible = {
			custom_tooltip = {
				fail_text = "must_build_around_star"
				is_star = yes
			}
		}
	}
	prerequisites = { tech_daedalusbrain }
	show_prereqs = yes
	build_time = 720
	resources = {
		category = megastructures
		cost = {
			im_infinityenergy = 1500
			im_infinitymatter = 285
		}
		upkeep = {
			im_infinitypoints = 50
			im_infinitymatter = 80
			energy = 100
		}
		produces = {
			im_infinityenergy = 500
		}
	}
	construction_blocks_and_blocked_by = self_type
	on_build_complete = {
		FROM = { set_country_flag = im_has_daedalusbrain }
		every_system_planet = {
			limit = { is_star = no }
			remove_planet = yes
		}
		set_asteroid_belt = {
			radius = 0
		}
	}
}
