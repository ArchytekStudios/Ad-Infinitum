can_destroy_planet_with_IM_ANGEL_SK = {
	custom_tooltip = {
		fail_text = is_not_star
		is_star = yes
	}
	custom_tooltip = {
		fail_text = is_black_hole
		NOT = { is_planet_class = pc_black_hole }
	}
	NOT = { is_planet_class = pc_im_searing_star }
	custom_tooltip = {
		fail_text = star_owner_is_not_hostile
		OR = {
			NOT = { exists = space_owner }
			space_owner = { is_hostile = from.owner }
			space_owner = { is_same_value = from.owner }
		}
	}
}

can_destroy_planet_with_IM_INF_RECONBEAM = {
	is_star = no
	is_asteroid = no
	custom_tooltip = {
		fail_text = "not_special_planets"
		NOR = {
			is_planet_class = pc_cosmogenesis_world
			im_is_seared_planet = yes
		}
	}
	custom_tooltip = {
		text = "not_hostile_obliterate"
		OR = {
			AND = {
				exists = space_owner
				space_owner = {
					OR = {
						is_at_war_with = from.owner
						is_country_type = swarm
						is_country_type = extradimensional
						is_country_type = extradimensional_2
						is_country_type = extradimensional_3
						is_country_type = ai_empire
					}
				}
			}
			AND = { # colony
				exists = owner
				owner = {
					OR = {
						is_at_war_with = from.owner
						is_country_type = swarm
						is_country_type = extradimensional
						is_country_type = extradimensional_2
						is_country_type = extradimensional_3
						is_country_type = ai_empire
					}
				}
			}
		}
	}
}
