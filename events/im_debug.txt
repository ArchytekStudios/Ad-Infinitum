country_event = {	# Toggle debug flag
	id = im_debug.1
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		if = {
			limit = { has_country_flag = im_debug_flag }
			remove_country_flag = im_debug_flag
		}
		else_if = {
			limit = { NOT = { has_country_flag = im_debug_flag } }
			set_country_flag = im_debug_flag
		}
	}
}
