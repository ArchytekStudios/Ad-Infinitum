namespace = im_debug

country_event = {	# Toggle debug flag
	id = im_debug.1
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		if = {
			limit = { has_country_flag = im_debug_flag }
			remove_country_flag = im_debug_flag
		}
		else_if = {
			limit = { NOT = { has_country_flag = im_debug_flag } }
			set_country_flag = im_debug_flag
		}
	}
}

country_event = {	# Debug event to check if the AP ticker is acting appropriately
	id = im_debug.2
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		OR = {
			has_nemesis = yes
			has_biogenesis_dlc = yes
			has_machine_age_dlc = yes
		}
		NOR = {
			has_ascension_perk = ap_become_the_crisis
			has_ascension_perk = ap_cosmogenesis
			has_ascension_perk = ap_defender_of_the_galaxy
			has_ascension_perk = ap_behemoths
			has_origin = origin_slavers # No crisis for MSI
			has_origin = origin_endbringers # No additional crisis for Endbringers
		}
		is_wilderness_empire = no

		OR = {
			has_technology = tech_psionic_theory
			has_technology = tech_zero_point_power
			has_technology = tech_positronic_implants
			has_technology = tech_sapient_ai
		}
	}
	immediate = {
	}
}
