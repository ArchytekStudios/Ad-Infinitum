namespace = im_general

system_event = {
	id = im_general.1995
	is_triggered_only = yes
	hide_window = yes
	immediate = {
		set_star_class = sc_im_searing_star
		every_system_planet = {
			IF = {
				limit = { 
					NOR = { 
						is_star = yes 
						clib_is_standard_star_planet = yes
						im_is_seared_planet = yes # can't re-sear a planet
					} 
				}
				kill_all_pop = yes
				destroy_colony = yes
				IF = { 
					limit = {
						NOT = { is_asteroid = yes }
						OR = {
							planet_size < 8
							is_moon = yes
						}
					} 
					change_pc = pc_im_obliterated_world 
				}
				ELSE_IF = { limit = { is_asteroid = yes } remove_planet = yes }
				ELSE = {
					SWITCH = { # cheers to Exakan on the modding den discord for this optimization
						trigger = is_planet_class
						pc_ringworld_habitable = { change_pc = pc_ringworld_habitable_damaged spawn_ringworld_cracker_effect = yes }
						pc_ringworld_shielded = { change_pc = pc_ringworld_habitable_damaged spawn_ringworld_cracker_effect = yes }
						pc_ringworld_seam = { change_pc = pc_ringworld_seam_damaged spawn_ringworld_cracker_effect = yes }
						pc_ringworld_tech = { change_pc = pc_ringworld_tech_damaged spawn_ringworld_cracker_effect = yes }
						pc_gas_giant = { change_pc = pc_molten change_planet_size = 5 }
						pc_habitat = { remove_planet = yes spawn_habitat_cracker_effect = yes }
						default = { change_pc = pc_im_bleeding_world }
					}
				}
			}
		}
		set_asteroid_belt = { radius = 0 }
		remove_star_flag = has_habitat
		every_system_megastructure = {
			limit = { NOT = { im_is_angelic_megastructure = yes } }
			SWITCH = {
				trigger = is_megastructure_type
				orbital_arc_furnace_destroyed = { remove_megastructure = this }
				cosmogenesis_world = { remove_megastructure = this spawn_lathe_cracter_effect = yes } #?todo: figure out how to replace with a ruined lathe
			}
			set_ruined_megastructure = yes
		}
	}
	after = {
		every_system_planet = { # remember to add more flags as necessary
			remove_planet_flag = has_arc_furnace
			SWITCH = {
				trigger = is_planet_class
				pc_im_obliterated_world = { add_deposit = d_im_obliterated_base }
				pc_im_bleeding_world = { add_deposit = d_im_bleeding_base }
			}
		}
	}
}
