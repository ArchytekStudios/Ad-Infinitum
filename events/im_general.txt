system_event = {
	id = im_general.1995
	is_triggered_only = yes
	hide_window = yes
	immediate = {
		set_star_class = sc_im_searing_star
		every_system_planet = {
			IF = {
				limit = { NOR = { is_star = yes clib_is_standard_star_planet = yes } }
				IF = { 
					limit = { OR = { clib_is_tiny_planet = yes is_planet_class = pc_gas_giant } }
					remove_planet = yes 
				}
				ELSE = { change_pc = pc_im_bleeding_world }
			}
		}
	}
}
